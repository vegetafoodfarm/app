คำแนะนำการเชื่อมต่อ Google Sheet:

  1. เปิด Google Sheet ชื่อ act ของคุณ
  2. ไปที่ ส่วนขยาย (Extensions) > Apps Script
  3. วางโค้ดด้านล่างนี้ลงไป แล้วกด บันทึก (Save)
  4. กด ทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้ใหม่ (New deployment)
  5. เลือกประเภทเป็น เว็บแอป (Web App) และตั้งค่า "ใครก็ได้ (Anyone)" เข้าถึงได้
  6. คัดลอก URL นำมาวางในช่องด้านบน
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("act");
  var data = JSON.parse(e.postData.contents);
  
  sheet.appendRow([
    data.date_formatted,  // A: date (DD/MM/YYYY)
    data.date_full,       // B: date (DD/MM/YYYY HH:mm)
    data.cate,            // C: cate
    data.by,              // D: by
    data.bed,             // E: bed
    data.qty,             // F: qty
    data.type || "",      // G: type
    data.tank_no || "",   // H: tank no
    data.age || "",       // I: age
    data.fert_bef || "",  // J: fert_bef
    data.fert_aft || "",  // K: fert_aft
    data.ph || "",        // L: ph
    data.level_bef || "", // M: level_bef
    data.level_aft || ""  // N: level_aft
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
